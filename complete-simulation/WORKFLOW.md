                    ┌─────────────────┐
                    │   温度场 T(x,t) │
                    │                 │
                    │  ∂T/∂t = α∇²T  │
                    │  + Q/(ρc)      │
                    └────────┬────────┘
                             │ 影响材料属性
                             ▼
        ┌───────────────────────────────────────────┐
        │                                           │
        │  E(T) = E₀[1 - β(T - T₀)]  (杨氏模量)    │
        │  α(T) = α₀ + α₁(T - T₀)    (热膨胀系数)  │
        │                                           │
        └───────────────────┬───────────────────────┘
                            │
                            ▼
┌─────────────────────────────────────────────────────────────────────────────┐
│                        位移场 u(x,t)                                        │
│                                                                             │
│   ∇·σ + ρb = ρü                                                             │
│                                                                             │
│   σ = C(T) : (ε - εᵗʰ)                                                      │
│                                                                             │
│   εᵗʰ = α(T)(T - T₀)I    ← 热应变                                           │
│                                                                             │

    │
    ├── [检查依赖]

    │
    ├── [检查输出]
    │   ├── 检查 Exodus 文件
    │   └── 检查 CSV 文件
    │
    ├── [运行后处理]
    │   └── python3 postprocess.py
    │       ├── 读取 CSV 数据

    │       └── 输出报告

    └── [显示结果]
